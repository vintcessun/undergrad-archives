## invert4.c
- 作用: 演示位运算在整数的低位反转中的应用，通过异或掩码实现指定数量比特位的翻转。  
- 功能: 用户输入整数，程序显示对应的二进制，然后反转低4位并展示改变后的结果。  
- 演示效果:
```c
/* invert4.c -- using bit operations to display binary */
#include <stdio.h>
#include <limits.h>
char * itobs(int, char *);
void show_bstr(const char *);
int invert_end(int num, int bits);

int main(void)
{
    char bin_str[CHAR_BIT * sizeof(int) + 1];
    
    int number;
    
    puts("Enter integers and see them in binary.");
    puts("Non-numeric input terminates program.");
    while (scanf("%d", &number) == 1)
    {
        itobs(number,bin_str);
        printf("%d is\n", number);
        show_bstr(bin_str);
        putchar('\n');
        number = invert_end(number, 4);
        printf("Inverting the last 4 bits gives\n");
        show_bstr(itobs(number,bin_str));
        putchar('\n');
    }
    puts("Bye!");
    
    return 0;
}

char * itobs(int n, char * ps)
{
    int i;
    const static int size = CHAR_BIT * sizeof(int);
    
    for (i = size - 1; i >= 0; i--, n >>= 1)
        ps[i] = (01 & n) + '0';
    ps[size] = '\0';
    
    return ps;
}

/* show binary string in blocks of 4 */
void show_bstr(const char * str)
{
    int i = 0;
	
    while (str[i])  /* not the null character */
    {
        putchar(str[i]);
        if(++i % 4 == 0 && str[i])
            putchar(' ');
    }
}

int invert_end(int num, int bits)
{
    int mask = 0;
    int bitval = 1;
    
    while (bits-- > 0)
    {
        mask |= bitval;
        bitval <<= 1;
    }
    
    return num ^ mask;
}
```

## fields.c
- 作用: 演示位域在结构体中的定义与使用，展示字段压缩存储的效果。  
- 功能: 定义包含位域的结构体，显示其大小与字段值，说明位域如何节省空间。  
- 演示效果:
```c
/* fields.c -- defining and using bit fields */
#include <stdio.h>
#include <stdbool.h>

struct {
    unsigned int autofil : 1;
    unsigned int bps : 4;
    unsigned int ccs : 4;
    unsigned int : 4;
    bool status : 1;
} printer;

int main(void)
{
    printf("sizeof printer = %zu\n", sizeof(printer));
    printer.autofil = 1;
    printer.bps = 9;
    printer.ccs = 12;
    printer.status = true;
    printf("autofil = %u\n", printer.autofil);
    printf("bps = %u\n", printer.bps);
    printf("ccs = %u\n", printer.ccs);
    printf("status = %d\n", printer.status);
    return 0;
}
```

```bash
gcc fields.c -o fields && echo "abc" | ./fields

autofil = 1
bps = 9
ccs = 12
status = 1
```

## dualview.c
- 作用: 演示位域与位运算结合，通过联合体实现同一数据的不同视图访问。  
- 功能: 用户输入整数，程序显示其二进制表示，并通过位域结构访问各部分位值。  
- 演示效果:
```c
/* dualview.c -- bit fields and bitwise operators */
#include <stdio.h>
#include <limits.h> // for CHAR_BIT
char * itobs(int, char *);
void show_bstr(const char *);

struct views {
    unsigned int : 6;
    unsigned int field1 : 3;
    unsigned int field2 : 3;
    unsigned int field3 : 3;
    unsigned int : 1;
};

union {
    struct views bf;
    int value;
} view;

int main(void)
{
    char bin_str[CHAR_BIT * sizeof(int) + 1];
    int number;
    
    puts("Enter integers and see them in binary.");
    puts("Non-numeric input terminates program.");
    while (scanf("%d", &number) == 1)
    {
        itobs(number, bin_str);
        printf("%d is ", number);
        show_bstr(bin_str);
        putchar('\n');
        view.value = number;
        printf("field1 = %d, field2 = %d, field3 = %d\n",
               view.bf.field1, view.bf.field2, view.bf.field3);
    }
    puts("Bye!");
    
    return 0;
}

    {
        putchar(str[i]);
        if (++i % 4 == 0 && str[i])
            putchar(' ');
    }
}
```

## align.c
- 作用: 演示 C11 中 `_Alignof` 与 `_Alignas` 的使用方式，说明类型和变量在内存中的对齐要求。  
- 功能: 输出不同类型的对齐要求，并显示各变量的内存地址。  
- 演示效果:
```c
//  align.c -- using _Alignof and _Alignas  (C11)

#include <stdio.h>
int main(void)
{
    double dx;
    char ca;
    char cx;
    double dz;
    char cb;
    char _Alignas(double)  cz;
   
    printf("char alignment:   %zd\n", _Alignof(char));
    printf("double alignment: %zd\n", _Alignof(double));
    printf("&dx: %p\n", &dx);
    printf("&ca: %p\n", &ca);
    printf("&cx: %p\n", &cx);
    printf("&dz: %p\n", &dz);
    printf("&cb: %p\n", &cb);
    printf("&cz: %p\n", &cz);
    
    return 0;
}
```
