#include <stdio.h>

{
    const double ANSWER = 3.14159;
    double response;
    
    printf("What is the value of pi?\n");
    scanf("%lf", &response);
    while (fabs(response - ANSWER) > 0.0001)

        printf("Try again!\n");

## truth.c
- 作用: 演示 C 语言中整数在布尔条件中的求值方式，非零为真，零为假
- 功能: 使用 while 循环测试不同整数值的真值性，展示正数、负数在条件判断中的效果
- 语义错误: 无
- 演示效果:
```c
// truth.c -- what values are true?
#include <stdio.h>
int main(void)
{
    int n = 3;
    
    while (n)
        printf("%2d is true\n", n--);
    printf("%2d is false\n", n);
    
    n = -3;
    while (n)
        printf("%2d is true\n", n++);
    printf("%2d is false\n", n);
    
    return 0;
}
```
```bash
3 is true
 2 is true
 1 is true
 0 is false
-3 is true
-2 is true
-1 is true
 0 is false
```

## when.c
- 作用: 演示 `while` 循环的退出条件，以及循环变量在循环中的变化过程
- 功能: 从 n=5 开始，当 n 小于 7 时进入循环，每次循环先输出当前值，再递增 n 并输出更新后的值，直到条件为假退出
- 语义错误: 无
- 演示效果:
```c
// when.c -- when a loop quits
#include <stdio.h>
int main(void)
{
    int n = 5;
    
    while (n < 7)                    // line 7
    {
        printf("n = %d\n", n);
        n++;                         // line 10
        printf("Now n = %d\n", n);   // line 11
    }
    printf("The loop has finished.\n");
    
    return 0;
}
```
```bash
n = 5
Now n = 6
n = 6
Now n = 7
The loop has finished.
```

## while1.c
- 作用: 演示 `while` 循环中使用花括号以防止无限循环
- 功能: 从 n=0 开始，当 n 小于 3 时进入循环，打印当前值并递增，循环结束后打印完成信息
- 主要错误: 原代码中缺少花括号，导致递增语句在循环外执行，形成死循环
- 演示效果:
```c
/* while1.c -- watch your braces       */
/* bad coding creates an infinite loop */
#include <stdio.h>
int main(void)
{
    int n = 0;
    
    while (n < 3)
    {
        printf("n is %d\n", n);
        n++;
    }
    printf("That's all this program does\n");
    
    return 0;
}
```
```bash
n is 0
n is 1
n is 2
That's all this program does
```

## while2.c
- 作用: 演示 `while` 循环语句中分号的影响
- 功能: 使用 `while (n++ < 3);` 语句实际运行了空循环体，循环结束后再执行 printf 打印 n 的值
- 语义错误: 分号导致循环体为空，n 在循环中递增至 4，循环结束后才打印结果
- 演示效果:
```c
/* while2.c -- watch your semicolons */
#include <stdio.h>
int main(void)
{
    int n = 0;
    
    while (n++ < 3);             /* line 7 */
    printf("n is %d\n", n);  /* line 8 */
    printf("That's all this program does.\n");
    
    return 0;
}
```
```bash
n is 4
That's all this program does.
```

## zeno.c
- 作用: 演示循环累加计算序列级数（Zeno’s paradox）的逼近行为
- 功能: 对输入的项数执行循环，每次将 1 除以 2 的幂次累加，展示时间趋近极限的过程
- 语义错误: 无
- 演示效果:
```c
/* zeno.c -- series sum */
#include <stdio.h>

int main(void)
{
    int t_ct;       // term count
    double time, power_of_2;
    int limit;
    
    printf("Enter the number of terms you want: ");
    scanf("%d", &limit);
    for (time=0, power_of_2=1, t_ct=1; t_ct <= limit;
                            t_ct++, power_of_2 *= 2.0)
    {
        time += 1.0/power_of_2;
        printf("time = %f when terms = %d.\n", time, t_ct);
    }
    
    return 0;
}
```
```bash
Enter the number of terms you want: time = 1.000000 when terms = 1.
time = 1.500000 when terms = 2.
time = 1.750000 when terms = 3.
time = 1.875000 when terms = 4.
time = 1.937500 when terms = 5.
```

- 功能: 循环读取用户输入的整数并累加，直到输入非数字（例如 q）结束

```c
// trouble.c -- misuse of =
// will cause infinite loop
#include <stdio.h>
int main(void)
{
    long num;
    long sum = 0L;
    int status;
    
    printf("Please enter an integer to be summed ");
    printf("(q to quit): ");
    status = scanf("%ld", &num);
    while (status == 1)
    {
        sum = sum + num;
        printf("Please enter next integer (q to quit): ");
        status = scanf("%ld", &num);
    }
    printf("Those integers sum to %ld.\n", sum);
    
    return 0;
}
```
```bash
Please enter an integer to be summed (q to quit): Please enter next integer (q to quit): Please enter next integer (q to quit): Please enter next integer (q to quit): Those integers sum to 6.
```

- 作用: 演示 `while` 循环的入口条件用法，在满足条件前反复执行循环体
- 功能: 循环提示用户输入秘密代码，若错误则继续提示直到输入正确
- 语义错误: 无
- 演示效果:
```c
/* entry.c -- entry condition loop */
#include <stdio.h>
int main(void)
{
    const int secret_code = 13;
    int code_entered;
    
    printf("To enter the triskaidekaphobia therapy club,\n");
    printf("please enter the secret code number: ");
    scanf("%d", &code_entered);
    while (code_entered != secret_code)
    {
        printf("To enter the triskaidekaphobia therapy club,\n");
        printf("please enter the secret code number: ");
        scanf("%d", &code_entered);
    }
    printf("Congratulations! You are cured!\n");
    
    return 0;
}
```

To enter the triskaidekaphobia therapy club,
